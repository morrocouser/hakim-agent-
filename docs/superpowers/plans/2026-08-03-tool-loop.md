# Registre d'outils + Tool Loop Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Exposer les 4 outils existants d'APEX (web_search, file_manager, code_executor, error_log) via un registre à décorateur `@axiom_tool`, branché sur le function calling natif OpenAI, avec une boucle interne optionnelle (`AXIOM_TOOL_LOOP=1`) dans `ReActLoop` et deux routes API (`GET /api/v1/tools`, `POST /api/v1/tools/invoke`).

**Architecture:** Un registre central (`tools/registry.py`) enregistre les outils via décorateur, génère les JSON-Schema (Pydantic), et fournit `build_tools_spec()` (format OpenAI) + `invoke_tool()`. Des wrappers (`tools/tool_wrappers.py`) enveloppent les helpers existants **sans les modifier**. `claude/client.py` gagne `chat_with_tools()` (function calling natif, `chat()` inchangée). `core/tool_loop.py` implémente la boucle interne plafonnée. `ReActLoop` l'active si le flag env `AXIOM_TOOL_LOOP=1` (défaut off). Deux routes exposent le registre.

**Tech Stack:** Python 3.14, FastAPI, Pydantic v2, httpx, venv `backend\.venv`. Tests = scripts autonomes assert sous `backend\tests\` lancés avec le python du venv (pas de pytest installé).

```python
from __future__ import annotations

import inspect
from dataclasses import dataclass
from typing import Any, Callable, Optional
from pydantic import BaseModel


@dataclass
class ToolContext:
    """Contexte d'exécution transmis à tous les outils.

    read_only=False : les outils en écriture sont autorisés.
    read_only=True  : tout outil en écriture renvoie une erreur (aucun prompt).
    """
    read_only: bool = False
    web_search: Any = None
    file_manager: Any = None
    code_executor: Any = None
    error_log: Any = None
    memory: Any = None


@dataclass
class ToolResult:
    """Résultat d'un outil. ``text`` est inséré tel quel dans la conversation."""
    ok: bool = True
    text: str = ""
    error: str = ""


_REGISTRY: dict[str, dict] = {}


def axiom_tool(
    name: str,
    description: str,
    permissions: str = "read",
    schema: Optional[type[BaseModel]] = None,
    available: Optional[Callable[[ToolContext], bool]] = None,
) -> Callable:
    """Décorateur : enregistre un outil dans le registre central."""
    def decorator(handler: Callable) -> Callable:
        _REGISTRY[name] = {
            "name": name,
            "description": description,
            "permissions": permissions,   # "read" | "write"
            "schema": schema,
            "available": available,
            "handler": handler,
        }
        return handler
    return decorator


def get_tools() -> list[dict]:
    """Retourne les métadonnées (sans handler) de tous les outils enregistrés."""
    return [
        {
            "name": t["name"],
            "description": t["description"],
            "permissions": t["permissions"],
            "parameters": t["schema"].model_json_schema() if t["schema"] else {"type": "object", "properties": {}},
        }
        for t in _REGISTRY.values()
    ]


def build_tools_spec(ctx: Optional[ToolContext] = None) -> list[dict]:
    """Construit la liste de tools au format OpenAI (function calling).

    Les outils dont ``available`` est défini et rend False pour un contexte
    donné sont exclus. Les outils en écriture restent présents même en
    ``read_only`` (c'est ``invoke_tool`` qui les évalue au moment de l'appel).
    """
    spec = []
    for meta in _REGISTRY.values():
        if meta.get("available") and ctx and not meta["available"](ctx):
            continue
        schema = meta["schema"].model_json_schema() if meta["schema"] else {"type": "object", "properties": {}}
        spec.append({
            "type": "function",
            "function": {
                "name": meta["name"],
                "description": meta["description"],
                "parameters": schema,
            },
        })
    return spec


async def invoke_tool(name: str, params: dict, ctx: Optional[ToolContext] = None) -> ToolResult:
    """Invoque un outil enregistré, avec validation Pydantic et permission."""
    ctx = ctx or ToolContext()
    meta = _REGISTRY.get(name)
    if not meta:
        return ToolResult(ok=False, error=f"Outil inconnu: {name}")
    if ctx.read_only and meta["permissions"] == "write":
        return ToolResult(ok=False, error=f"Outil en écriture bloqué (lecture seule): {name}")
    if meta.get("available") and not meta["available"](ctx):
        return ToolResult(ok=False, error=f"Outil indisponible: {name}")
    try:
        if meta["schema"] is not None:
            validated = meta["schema"].model_validate(params or {})
        else:
            validated = params
        result = meta["handler"](validated, ctx)
        if inspect.isawaitable(result):
            result = await result
        return ToolResult(ok=True, text=str(result))
    except Exception as e:
        return ToolResult(ok=False, error=f"[Tool {name}] {e}")
```

```python
    async def chat_with_tools(
        self,
        model: str,
        system: str,
        messages: list[dict],
        tools: Optional[list[dict]] = None,
        max_tokens: int = 4096,
    ) -> dict:
        """Function calling natif OpenAI ; `chat()` reste inchangé.

        Retourne {"content": str, "tool_calls": list[dict], "error": Optional[str]}.
        `tool_calls[i]` = {"id", "name", "arguments"} avec arguments déjà désérialisé.
        """
        model = os.getenv("AXIOM_LLM_MODEL") or model
        estimated_tokens = max_tokens + len(system) // 4 + sum(len(m.get("content", "")) for m in messages) // 4
        await self.rate_limiter.acquire(estimated_tokens)

        body = {
            "model": model,
            "messages": [{"role": "system", "content": system}] + messages,
            "max_tokens": max_tokens,
        }
        if tools:
            body["tools"] = tools
            body["tool_choice"] = "auto"

        headers = {
            "Content-Type": "application/json",
            "HTTP-Referer": APP_REFERRER,
            "X-Title": APP_TITLE,
        }
        if self._api_key:
            headers["Authorization"] = f"Bearer {self._api_key}"

        try:
            resp = await self.http.post(
                f"{self.base_url}/chat/completions",
                headers=headers,
                json=body,
            )
            text = resp.text
            if "data: [DONE]" in text:
                text = text.split("data: [DONE]")[0].strip()
            start_idx = text.find("{")
            end_idx = text.rfind("}")
            if start_idx != -1 and end_idx != -1:
                text = text[start_idx : end_idx + 1]
            data = json.loads(text)
            if resp.is_error:
                err = data.get("error", {}).get("message", str(resp.status_code))
                return {"content": "", "tool_calls": [], "error": f"[Tool API] {err}"}
            message = data["choices"][0]["message"]
            content = message.get("content") or ""
            calls = []
            for tc in message.get("tool_calls") or []:
                fn = tc.get("function") or {}
                try:
                    arguments = json.loads(fn.get("arguments") or "{}")
                except Exception:
                    arguments = {}
                calls.append({
                    "id": tc.get("id", ""),
                    "name": fn.get("name", ""),
                    "arguments": arguments,
                })
            return {"content": content, "tool_calls": calls, "error": None}
        except Exception as e:
            return {"content": "", "tool_calls": [], "error": f"[Tool API] {str(e)}"}
```

```python
from __future__ import annotations

import json
from typing import Optional

from axiom.tools.registry import ToolContext, build_tools_spec, invoke_tool

DEFAULT_MAX_STEPS = 5
MAX_MESSAGES = 24


def truncate_messages(messages: list[dict]) -> list[dict]:
    """Garder le premier message utilisateur + les derniers MAX_MESSAGES."""
    if len(messages) <= MAX_MESSAGES:
        return messages
    head = [m for m in messages if m.get("role") == "user"][:1]
    tail = messages[1:]
    limit = MAX_MESSAGES - len(head)
    return head + tail[-limit:]


def _assistant_with_tool_calls(resp: dict, calls: list[dict]) -> dict:
    return {
        "role": "assistant",
        "content": resp.get("content") or "",
        "tool_calls": [
            {
                "id": c["id"],
                "type": "function",
                "function": {"name": c["name"], "arguments": json.dumps(c["arguments"], ensure_ascii=False)},
            }
            for c in calls
        ],
    }


async def run_tool_loop(
    claude,
    model: str,
    system: str,
    user_prompt: str,
    ctx: Optional[ToolContext] = None,
    max_steps: int = DEFAULT_MAX_STEPS,
) -> str:
    """Boucle interne de function calling. Retourne le texte final.

    Si aucune tool n'est exposée (spec vide) ou si le provider refuse le
    function calling (erreur), on replie sur un appel `chat()` simple.
    """
    ctx = ctx or ToolContext()
    tools = build_tools_spec(ctx)
    messages = [{"role": "user", "content": user_prompt}]

    if not tools:
        return await claude.chat(model=model, system=system, messages=messages)

    for _ in range(max_steps):
        resp = await claude.chat_with_tools(
            model=model, system=system, messages=messages, tools=tools,
        )
        if resp.get("error"):
            # Repli : provider refuse/dysfonctionne → chat simple sans tools
            return await claude.chat(model=model, system=system, messages=messages)

        calls = resp.get("tool_calls") or []
        if not calls:
            return resp.get("content") or ""

        messages.append(_assistant_with_tool_calls(resp, calls))
        for call in calls:
            result = await invoke_tool(call["name"], call["arguments"], ctx)
            if result.ok:
                content = result.text
            else:
                content = f"[Tool {call['name']} error] {result.error}"
            messages.append({"role": "tool", "tool_call_id": call["id"], "content": content})
        messages = truncate_messages(messages)

    # Plafond de steps atteint : retour du dernier contenu assistant produit
    last = messages[-1]
    return last.get("content") or "Boucle d'outils plafonnée (max_steps atteint)."
```
